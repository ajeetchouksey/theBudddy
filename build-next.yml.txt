# Push to main: validate with main rules, build twice (must be identical), upload artefacts.
# Publishing to the next channel is a maintainer TODO; the release workflow is task P11.
# TODO(maintainers): add a publish job bound to a "next" environment (test Drive folder,
# latest-next.json on GitHub Pages) once the Drive upload credential exists.
# See docs/release-management.md and docs/tech-stack-development.md.
name: build-next

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions: read-all

concurrency:
  group: build-next
  cancel-in-progress: false

jobs:
  build:
    name: build
    runs-on: ubuntu-latest
    timeout-minutes: 15
    permissions:
      contents: read
    env:
      GSC_BRANCH: main
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          persist-credentials: false
      - uses: actions/setup-python@5fda3b95a4ea91299a34e894583c3862153e4b97 # v7.0.0
        with:
          python-version: "3.12"
          cache: pip
          cache-dependency-path: requirements-dev.txt
      - name: Install tools
        run: python -m pip install -r requirements-dev.txt
      - name: Validate with main-branch rules
        run: python scripts/validate.py
      - name: Build twice and compare
        run: |
          SOURCE_DATE_EPOCH="$(git log -1 --format=%ct)"
          export SOURCE_DATE_EPOCH
          python scripts/build.py
          mkdir -p "$RUNNER_TEMP/first"
          cp -r build skill/reference/packs "$RUNNER_TEMP/first/"
          python scripts/build.py
          diff -r "$RUNNER_TEMP/first/build" build
          diff -r "$RUNNER_TEMP/first/packs" skill/reference/packs
      - name: Checksums
        working-directory: build
        run: find . -type f ! -name README.md ! -name SHA256SUMS -print0 | sort -z | xargs -0 sha256sum > SHA256SUMS
      - uses: actions/upload-artifact@043fb46d1a93c77aae656e7c1c64a875d1fc6a0a # v7.0.1
        with:
          name: next-${{ github.sha }}
          path: |
            build/
            skill/reference/packs/
          if-no-files-found: error
          retention-days: 30
